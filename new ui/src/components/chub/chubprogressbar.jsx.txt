import React from "react";

export default function CHubProgressBar({ value = 0, max = 100 }) {
  const pct = Math.min(100, Math.max(0, (value / max) * 100));
  return (
    <div className="h-[5px] w-full bg-[#080B0F] shadow-[inset_0_1px_2px_rgba(0,0,0,0.9)] ring-1 ring-inset ring-[#1B2028]">
      <div
        className="h-full bg-gradient-to-r from-[#8C5A17] via-[#D9A02A] to-[#F0C25A] shadow-[0_0_8px_rgba(217,160,42,0.45)] transition-all duration-300"
        style={{ width: `${pct}%` }}
      />
    </div>
  );
}
